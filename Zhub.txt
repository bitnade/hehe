local FreeCamPart, freecamConnection
local function ToggleFreeCam()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")

    if _G.FreeCam then
    if hrp then hrp.Anchored = true end
    if not FreeCamPart then
        FreeCamPart = Instance.new("Part")
        FreeCamPart.Anchored = true
        FreeCamPart.CanCollide = false
        FreeCamPart.Transparency = 1
        FreeCamPart.CFrame = Camera.CFrame
        FreeCamPart.Parent = workspace
    end
    Camera.CameraSubject = FreeCamPart

    freecamConnection = RunService.RenderStepped:Connect(function(dt)
        local dir = Vector3.new()
        local hum = char and char:FindFirstChild("Humanoid")
        if hum and hum.MoveDirection.Magnitude > 0 then
            dir = hum.MoveDirection
        else
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + Camera.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - Camera.CFrame.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - Camera.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + Camera.CFrame.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.E) then dir = dir + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.Q) then dir = dir - Vector3.new(0, 1, 0) end
        end

        local camSpeed = _G.SpeedEnabled and _G.MovementSpeed or 50
        if dir.Magnitude > 0 then
            FreeCamPart.CFrame = FreeCamPart.CFrame + (dir.Unit * (camSpeed * dt))
        end
    end)
    else
    if hrp then hrp.Anchored = false end
    if freecamConnection then freecamConnection:Disconnect() end
    if FreeCamPart then FreeCamPart:Destroy(); FreeCamPart = nil end
    if char and char:FindFirstChild("Humanoid") then
        Camera.CameraSubject = char.Humanoid
    end
    end
end

-- [[ KILL AURA SYSTEM ]] --
-- Tracks last-hit time per NPC to throttle damage ticks
local KillAuraLastHit = {}
-- How many times per second kill aura deals damage
local KILL_AURA_RATE = 0.1 -- seconds between damage applications per NPC

--[[
    IsPlayerCharacter: returns true if a model belongs to a real player.
    Uses GetPlayerFromCharacter (O(1)) instead of looping all players.
]]
local function IsPlayerCharacter(model)
    return Players:GetPlayerFromCharacter(model) ~= nil
end

--[[
    GetNPCsInRange: scans workspace for models with a Humanoid and a
    HumanoidRootPart that are (a) not a player character, (b) alive,
    and (c) within _G.KillAuraRange studs of our HRP.
    OPTIMIZED: caches scan results for 0.5 seconds to avoid
    calling workspace:GetDescendants() every frame.
]]
local _npcCache = {}
local _npcCacheTime = 0
local NPC_CACHE_DURATION = 0.5 -- rescan every 0.5 seconds

local function GetNPCsInRange(hrp)
    local results = {}
    local range = _G.KillAuraRange
    local now = tick()

    -- Refresh NPC list only every NPC_CACHE_DURATION seconds
    if (now - _npcCacheTime) >= NPC_CACHE_DURATION then
    _npcCacheTime = now
    _npcCache = {}

    local function ScanFolder(parent, depth)
        if (depth or 0) > 3 then return end -- max folder depth 3
        for _, child in pairs(parent:GetChildren()) do
            if child:IsA("Model") and not IsPlayerCharacter(child) then
                local hum = child:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    local root = child:FindFirstChild("HumanoidRootPart")
                        or child:FindFirstChild("Torso")
                        or child:FindFirstChildWhichIsA("BasePart")
                    if root then
                        table.insert(_npcCache, {humanoid = hum, root = root})
                    end
                end
            elseif child:IsA("Folder") or child:IsA("Model") or child.Name == "NPCs" or child.Name == "Enemies" or child.Name == "Monsters" or child.Name == "Bots" then
                ScanFolder(child, (depth or 0) + 1)
            end
        end
    end

    ScanFolder(workspace, 0)
    end

    -- Filter cached NPCs by range
    for _, entry in pairs(_npcCache) do
    if entry.humanoid and entry.humanoid.Parent and entry.humanoid.Health > 0 then
        local dist = (entry.root.Position - hrp.Position).Magnitude
        if dist <= range then
            table.insert(results, entry)
        end
    end
    end

    return results
end

--[[
    RunKillAura: called every Stepped tick.
    Applies damage to every NPC in range, rate-limited per NPC so we
    don't spam TakeDamage every frame (which can cause issues).
    Damage method priority:
      1. TakeDamage (standard, works on most games)
      2. Fires the tool's activation if the player has a damage tool equipped
     (fallback for games that block TakeDamage via HealthChanged guards)
]]
local function RunKillAura()
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local now = tick()
    local npcs = GetNPCsInRange(hrp)

    for _, entry in pairs(npcs) do
    local hum = entry.humanoid
    local key = tostring(hum)

    -- Rate-limit: only damage each NPC every KILL_AURA_RATE seconds
    if not KillAuraLastHit[key] or (now - KillAuraLastHit[key]) >= KILL_AURA_RATE then
        KillAuraLastHit[key] = now

        -- Primary method: TakeDamage
        local ok, err = pcall(function()
            hum:TakeDamage(math.huge)
        end)

        -- Fallback: if TakeDamage is blocked, try setting Health directly
        if not ok or hum.Health > 0 then
            pcall(function()
                hum.Health = 0
            end)
        end

        -- Secondary fallback: simulate tool hit via firetouchinterest on HRP
        if hum.Health > 0 then
            pcall(function()
                firetouchinterest(hrp, entry.root, 0)
                firetouchinterest(hrp, entry.root, 1)
            end)
        end
    end
    end

    -- Clean up stale keys so the table doesn't grow forever
    for key, lastTime in pairs(KillAuraLastHit) do
    if (now - lastTime) > 10 then
        KillAuraLastHit[key] = nil
    end
    end
end

local _lastClickTime = 0
local VIM = game:GetService("VirtualInputManager")

local function RunAutoClicker()
    if not _G.AutoClicker then return end
    local now = tick()
    if (now - _lastClickTime) < (_G.ClickInterval or 0.1) then return end
    _lastClickTime = now

    local centerPos = AutoClickerTarget.AbsolutePosition + (AutoClickerTarget.AbsoluteSize / 2)
    local posX, posY = centerPos.X, centerPos.Y

    pcall(function()
        if UserInputService.TouchEnabled then
            -- Dispatches native mobile Touch events (Touch Begin -> Touch End) without switching input modes!
            VIM:SendTouchEvent(9999, Enum.UserInputState.Begin, Vector2.new(posX, posY))
            task.delay(0.02, function()
                pcall(function()
                    VIM:SendTouchEvent(9999, Enum.UserInputState.End, Vector2.new(posX, posY))
                end)
            end)
        else
            -- PC Mouse Click fallback
            VIM:SendMouseButtonEvent(posX, posY, 0, true, game, 0)
            task.delay(0.02, function()
                pcall(function()
                    VIM:SendMouseButtonEvent(posX, posY, 0, false, game, 0)
                end)
            end)
        end
    end)
end

-- [[ WALL CHECK ]] --
local AimRayParams = RaycastParams.new()
AimRayParams.FilterType = Enum.RaycastFilterType.Exclude
AimRayParams.IgnoreWater = true

local function IsPartVisible(targetPart)
    if not _G.AimWallCheck then return true end
    local myChar = LocalPlayer.Character
    if not myChar or not targetPart then return false end

    local origin = Camera.CFrame.Position
    local direction = targetPart.Position - origin
    if direction.Magnitude < 0.5 then return true end

    -- Build exclude list once
    AimRayParams.FilterDescendantsInstances = {myChar, targetPart.Parent}

    local hit = workspace:Raycast(origin, direction, AimRayParams)
    if not hit then return true end

    local inst = hit.Instance
    if not inst then return true end

    -- Allow shots through: own character parts, non-collidable, transparent, accessories
    if inst:IsDescendantOf(targetPart.Parent) then return true end
    if inst.CanCollide == false then return true end
    if inst.Transparency >= 0.5 then return true end
    if inst:IsA("Accessory") or inst:FindFirstAncestorWhichIsA("Accessory") then return true end

    return false
end

-- [[ TARGET SELECTION ]] --
local CurrentLockedTarget = nil  -- The character Model we're locked onto
local CurrentAimIsHead = false   -- Whether this engagement aims at head height

-- [[ TEAMMATE / HARMABILITY CHECK ]] --
local function IsTeammate(player)
    if not _G.TeamCheck then return false end
    if not player or player == LocalPlayer then return false end

    -- Team check: same team = teammate
    if LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team then
    -- Both neutral means no team system, don't skip
    if not (LocalPlayer.Neutral and player.Neutral) then
        return true
    end
    end

    -- ForceField check: player has spawn protection
    local char = player.Character
    if char and char:FindFirstChildOfClass("ForceField") then
    return true
    end

    return false
end

local function FindBestTarget()
    local crosshairPos = GetCrosshairPosition()
    local myChar = LocalPlayer.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")

    -- Sticky lock: keep tracking same character if still valid (prevents target-hopping jitter)
    if CurrentLockedTarget and CurrentLockedTarget.Parent then
    local hum = CurrentLockedTarget:FindFirstChildOfClass("Humanoid")
    local player = Players:GetPlayerFromCharacter(CurrentLockedTarget)
    if player then
        local ign = _G.IgnoredPlayers[player.Name] or _G.IgnoredPlayers[player.DisplayName] or _G.IgnoredPlayers[tostring(player.UserId)]
        if not ign and not IsTeammate(player) and hum and hum.Health > 0 then
            local hrp = CurrentLockedTarget:FindFirstChild("HumanoidRootPart") or CurrentLockedTarget:FindFirstChild("Torso")
            if hrp then
                if myHRP and (hrp.Position - myHRP.Position).Magnitude <= _G.ESPRange then
                    local sp, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    if onScreen then
                        local d = (Vector2.new(sp.X, sp.Y) - crosshairPos).Magnitude
                        if d <= _G.AimFOV * 1.5 then
                            local head = CurrentLockedTarget:FindFirstChild("Head")
                            if IsPartVisible(hrp) or (head and IsPartVisible(head)) then
                                return CurrentLockedTarget
                            end
                        end
                    end
                end
            end
        end
    end
    end

    -- Find new closest target (FAST FILTER FIRST: screen position & FOV distance -> SLOW CHECK: Raycast)
    local isDotMode = (not _G.AutoAim) and _G.AutoShoot
    local effectiveFOV = isDotMode and (UserInputService.TouchEnabled and 30 or 15) or _G.AimFOV

    local bestChar = nil
    local bestDist = effectiveFOV

    for _, p in pairs(Players:GetPlayers()) do
    if p ~= LocalPlayer then
        local char = p.Character
        if char then
            local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
            if hrp then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    local dist3D = myHRP and (hrp.Position - myHRP.Position).Magnitude or 0
                    if dist3D <= _G.ESPRange then
                        -- Fast Check 1: Is player on screen?
                        local sp, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                        if onScreen then
                            -- Fast Check 2: Is 2D distance to crosshair closer than bestDist?
                            local d = (Vector2.new(sp.X, sp.Y) - crosshairPos).Magnitude
                            if d < bestDist then
                                -- Fast Check 3: Is player ignored or teammate?
                                local ign = _G.IgnoredPlayers[p.Name] or _G.IgnoredPlayers[p.DisplayName] or _G.IgnoredPlayers[tostring(p.UserId)]
                                if not ign and not IsTeammate(p) then
                                    -- Slow Check 4: Wall check raycast ONLY for the 1 candidate inside FOV!
                                    local head = char:FindFirstChild("Head")
                                    if IsPartVisible(hrp) or (head and IsPartVisible(head)) then
                                        bestDist = d
                                        bestChar = char
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    end

    -- New target acquired — roll headshot chance once per engagement
    if bestChar ~= CurrentLockedTarget then
    CurrentLockedTarget = bestChar
    CurrentAimIsHead = bestChar and (math.random() <= (_G.HeadshotChance or 0.40))
    end

    return bestChar
end

-- [[ CAMERA TRACKING ]] --
local function RunAutoAim(dt)
    if not _G.AutoAim then
    CurrentLockedTarget = nil
    return
    end

    local targetChar = FindBestTarget()
    if not targetChar then
    CurrentLockedTarget = nil
    return
    end

    local hrp = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Torso")
    if not hrp then return end

    -- Stable aim point: HumanoidRootPart position + fixed Y offset (no animation jitter)
    local yOffset = CurrentAimIsHead and 1.5 or 0
    local aimPoint = hrp.Position + Vector3.new(0, yOffset, 0)

    -- Gentle velocity lead (only horizontal, avoids vertical overshoot from jumping)
    local vel = Vector3.new()
    pcall(function()
    local v = hrp.AssemblyLinearVelocity or hrp.Velocity
    if v then vel = Vector3.new(v.X, 0, v.Z) end
    end)
    aimPoint = aimPoint + vel * 0.02

    if _G.SmoothAim then
    -- Smooth Aim: lerp the camera towards the target using AimSmoothness
    local targetCFrame = CFrame.new(Camera.CFrame.Position, aimPoint)
    local alpha = math.clamp(_G.AimSmoothness * (dt * 60), 0, 1)
    Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, alpha)
    else
    -- Direct Aim: snap camera directly to target every frame (instant lock)
    Camera.CFrame = CFrame.new(Camera.CFrame.Position, aimPoint)
    end
end

-- [[ AUTO SHOOT SYSTEM ]] --
local _lastShootTime = 0
local AUTO_SHOOT_RATE = 0.1 -- 10 shots per second max

local function RunAutoShoot()
    if not _G.AutoShoot then return end
    if not _G.AutoAim then return end

    local targetChar = CurrentLockedTarget
    if not targetChar or not targetChar.Parent then
        targetChar = FindBestTarget()
    end
    if not targetChar or not targetChar.Parent then return end

    local now = tick()
    if (now - _lastShootTime) < AUTO_SHOOT_RATE then return end
    _lastShootTime = now

    local char = LocalPlayer.Character
    if not char then return end

    -- Fire weapon natively via Roblox Lua API without sending fake mouse clicks
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() tool:Activate() end)

        -- Trigger gun RemoteEvents inside tool for custom gun frameworks
        pcall(function()
            for _, child in ipairs(tool:GetDescendants()) do
                if child:IsA("RemoteEvent") then
                    local lname = string.lower(child.Name)
                    if lname:find("shoot") or lname:find("fire") or lname:find("remote") or lname:find("gun") or lname:find("action") or lname:find("bullet") or lname:find("attack") or lname:find("damage") then
                        pcall(function() child:FireServer() end)
                    end
                end
            end
        end)
    end
end

local function SetESPState(player, visible)
    if player == LocalPlayer then return end
    local char = player.Character
    if not char then return end

    local head = char:FindFirstChild("Head")
    local high = char:FindFirstChild("ZESP")
    local tag = char:FindFirstChild("ZTag") or (head and head:FindFirstChild("ZTag"))

    if visible then
    if not high or not tag then
        high, tag = GetOrCreateESP(player)
    end
    if high then high.Enabled = true end
    if tag then
        tag.MaxDistance = _G.ESPRange or 1000
        tag.Enabled = true
    end
    else
    if high then high.Enabled = false end
    if tag then tag.Enabled = false end
    end
end

-- ESP_RANGE is now controlled by _G.ESPRange (default 1000 studs)

local ESPRayParams = RaycastParams.new()
ESPRayParams.FilterType = Enum.RaycastFilterType.Exclude
ESPRayParams.IgnoreWater = true

local function IsPlayerOccluded(targetChar)
    local myChar = LocalPlayer.Character
    if not myChar or not targetChar then return false end

    local hrp = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("Head")
    if not hrp then return false end

    local origin = Camera.CFrame.Position
    local direction = hrp.Position - origin
    local dist = direction.Magnitude
    if dist < 0.5 or dist > _G.ESPRange then return false end

    -- Fast check: don't raycast for players off-screen
    local _, onScreen = Camera:WorldToViewportPoint(hrp.Position)
    if not onScreen then return false end

    ESPRayParams.FilterDescendantsInstances = {myChar, targetChar}

    local hit = workspace:Raycast(origin, direction, ESPRayParams)
    if not hit then return false end

    local inst = hit.Instance
    if not inst then return false end

    if inst:IsDescendantOf(targetChar) or inst:IsDescendantOf(myChar) then return false end
    if inst.CanCollide == false then return false end
    if inst.Transparency >= 0.5 then return false end
    if inst:IsA("Accessory") or inst:FindFirstAncestorWhichIsA("Accessory") then return false end

    return true
end

local function RefreshAllESP()
    local myChar = LocalPlayer.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")

    for _, p in pairs(Players:GetPlayers()) do
    if p ~= LocalPlayer then
        if _G.EspEnabled and myHRP then
            local theirChar = p.Character
            if theirChar then
                local theirHRP = theirChar:FindFirstChild("HumanoidRootPart") or theirChar:FindFirstChild("Torso")
                if theirHRP and (theirHRP.Position - myHRP.Position).Magnitude <= _G.ESPRange and IsPlayerOccluded(theirChar) then
                    SetESPState(p, true)
                else
                    SetESPState(p, false)
                end
            else
                SetESPState(p, false)
            end
        else
            SetESPState(p, false)
        end
    end
    end
end

-- [[ ZERO-ALLOCATION NOCLIP PART CACHE (PREVENTS GC FRAME DROPS) ]] --
local _noclipPartsList = {}
local _noclipCharTracked = nil
local _noclipAddedConn, _noclipRemovedConn

local function SetupNoclipCache(char)
_noclipPartsList = {}
if not char then return end
for _, v in ipairs(char:GetDescendants()) do
    if v:IsA("BasePart") then
    table.insert(_noclipPartsList, v)
    end
end
if _noclipAddedConn then _noclipAddedConn:Disconnect() _noclipAddedConn = nil end
if _noclipRemovedConn then _noclipRemovedConn:Disconnect() _noclipRemovedConn = nil end

_noclipAddedConn = char.DescendantAdded:Connect(function(v)
    if v:IsA("BasePart") then
    table.insert(_noclipPartsList, v)
    end
end)
_noclipRemovedConn = char.DescendantRemoving:Connect(function(v)
    if v:IsA("BasePart") then
    for i = #_noclipPartsList, 1, -1 do
        if _noclipPartsList[i] == v then
            table.remove(_noclipPartsList, i)
            break
        end
    end
    end
end)
end

    end
    else
    if _noclipCharTracked then
        _noclipCharTracked = nil
        _noclipPartsList = {}
        if _noclipAddedConn then _noclipAddedConn:Disconnect() _noclipAddedConn = nil end
        if _noclipRemovedConn then _noclipRemovedConn:Disconnect() _noclipRemovedConn = nil end
    end
    end



    -- Walk Speed (only modify WalkSpeed when Speed feature is active)
    if _G.SpeedEnabled and not _G.Fly then
    char.Humanoid.WalkSpeed = _G.MovementSpeed
    end

    -- Fly
    if _G.Fly then
    char.Humanoid.PlatformStand = true
    local hrp = char.HumanoidRootPart
    local bv = hrp:FindFirstChild("VelocityGrip") or Instance.new("BodyVelocity", hrp)
    local bg = hrp:FindFirstChild("GyroGrip")     or Instance.new("BodyGyro",     hrp)
    bv.Name = "VelocityGrip"
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bg.Name = "GyroGrip"
    bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)

    local dir = char.Humanoid.MoveDirection
    if dir.Magnitude == 0 then
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + Camera.CFrame.RightVector end
    end

    local flySpeed = _G.SpeedEnabled and _G.MovementSpeed or 20
    bv.Velocity = dir * flySpeed
    bg.CFrame = Camera.CFrame
    end

    -- Reach - Update tool hitboxes when equipped
    if _G.ReachEnabled then
    ExpandToolHitbox()
    UpdateReachSize()
    end
end
end)
